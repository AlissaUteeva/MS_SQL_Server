## 2.4 Надежность
После COMMIT оценка студента сохраняется даже при перезапуске SQL Server. Объясните роль журнала транзакций (LDF) на примере.

## 3.1 READ UNCOMMITTED (Грязное чтение)
Сценарий: Деканат вносит новую оценку, но ещё не закоммитил. Студент смотрит свою ведомость и видит «2», хотя преподаватель передумал и откатит.

-- Session 1 (деканат)
BEGIN TRAN;
INSERT INTO Grades (StudentId, CourseId, Grade) VALUES (1, 10, 2);
-- пауза

-- Session 2 (студент)
SELECT * FROM Grades WITH (NOLOCK) WHERE StudentId = 1; -- увидит 2

-- Session 1
ROLLBACK;  -- оценка исчезла, но студент её уже «увидел»

## 3.2 READ COMMITTED (Non-repeatable read)
Сценарий: Студент дважды читает свою среднюю оценку во время сессии. Между чтениями преподаватель выставляет новую оценку.

(```)sql
-- Session 1
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
BEGIN TRAN;
SELECT AVG(Grade) FROM Grades WHERE StudentId = 1;  -- например 4.0
-- пауза

-- Session 2
INSERT INTO Grades VALUES (1, 11, 5);  -- новая оценка

-- Session 1
SELECT AVG(Grade) FROM Grades WHERE StudentId = 1;  -- уже 4.3
COMMIT;

## 3.3 REPEATABLE READ (Phantom read)
Сценарий: Деканат формирует список студентов группы ИС-21. Пока он читает, приходит перевод нового студента в эту группу.

(```)sql
SET TRANSACTION ISOLATION LEVEL REPEATABLE READ;
BEGIN TRAN;
SELECT COUNT(*) FROM Students WHERE GroupId = 5;  -- 20
-- пауза, другая сессия добавляет студента в группу 5
SELECT COUNT(*) FROM Students WHERE GroupId = 5;  -- 21 → фантом
COMMIT;

## 3.4 SERIALIZABLE
Сценарий: Проверка, что в группе не больше 25 студентов. Запрос и вставка должны быть защищены от гонок.

SET TRANSACTION ISOLATION LEVEL SERIALIZABLE;
BEGIN TRAN;
IF (SELECT COUNT(*) FROM Students WHERE GroupId = 5) < 25
    INSERT INTO Students (FullName, GroupId, EnrolledYear) 
    VALUES (N'Иванов И.И.', 5, 2025);
COMMIT;

## 3.5 SNAPSHOT
Отчёт по успеваемости факультета должен читаться согласованно, без блокировки текущих операций выставления оценок.

ALTER DATABASE UniversityDB SET ALLOW_SNAPSHOT_ISOLATION ON;

SET TRANSACTION ISOLATION LEVEL SNAPSHOT;
BEGIN TRAN;
-- длинный аналитический отчёт, не блокирует вставку оценок
SELECT * FROM Grades g JOIN Students s ON g.StudentId = s.StudentId;
COMMIT;

COMMIT;

## 4.4 Эскалация блокировок
Сценарий: массовое выставление оценок всем 5000 студентам. SQL Server может эскалировать строковые блокировки в табличную. Продемонстрируйте через sys.dm_tran_locks.

## 4.5 Кто кого блокирует
SELECT 
    blocked.session_id AS Заблокированный,
    blocking.session_id AS Блокирующий,
    blocked.wait_type, blocked.wait_time,
    blocked.wait_resource
FROM sys.dm_exec_requests blocked
JOIN sys.dm_exec_sessions blocking 
    ON blocked.blocking_session_id = blocking.session_id
WHERE blocked.blocking_session_id <> 0;

Воспроизведите: одна сессия обновляет оценку, другая читает без hint — увидите блокировку.

## 5.1 Воспроизведение deadlock
Сценарий: Два преподавателя одновременно выставляют оценки двум студентам в разном порядке.

-- Session 1 (Преподаватель А)
BEGIN TRAN;
UPDATE Grades SET Grade = 5 WHERE StudentId = 1 AND CourseId = 10;
WAITFOR DELAY '00:00:05';
UPDATE Grades SET Grade = 4 WHERE StudentId = 2 AND CourseId = 10;
COMMIT;

-- Session 2 (Преподаватель Б)
BEGIN TRAN;
UPDATE Grades SET Grade = 4 WHERE StudentId = 2 AND CourseId = 10;
WAITFOR DELAY '00:00:05';
UPDATE Grades SET Grade = 5 WHERE StudentId = 1 AND CourseId = 10;
COMMIT;

Одна сессия станет жертвой deadlock (ошибка 1205).

## 7.2 Откат вложенной транзакции
Покажите, что ROLLBACK внутри вложенной транзакции откатывает всю транзакцию (включая внешнюю).
```
CREATE PROCEDURE usp_AddGrade @StudentId INT, @CourseId INT, @Grade INT
AS
BEGIN
    BEGIN TRAN;
        INSERT INTO Grades (StudentId, CourseId, Grade) 
        VALUES (@StudentId, @CourseId, @Grade);
    COMMIT;
END;

-- Внешний вызов
BEGIN TRAN;
EXEC usp_AddGrade 1, 10, 5;
SELECT @@TRANCOUNT;  -- 1, потому что внутренний COMMIT уменьшает лишь до внешнего
COMMIT;
```

## 7.3 Безопасный шаблон
Напишите процедуру usp_ExpelStudent (отчисление) с корректной обработкой вложенных транзакций:
```
CREATE PROCEDURE usp_ExpelStudent @StudentId INT
AS
BEGIN
    DECLARE @tranCount INT = @@TRANCOUNT;
    BEGIN TRY
        IF @tranCount = 0 BEGIN TRAN;

        UPDATE Students SET Status = 'Отчислен' WHERE StudentId = @StudentId;
        INSERT INTO AuditLog (EventType, TableName, RecordId, Message)
        VALUES ('Отчисление', 'Students', @StudentId, N'Студент отчислен');

        IF @tranCount = 0 COMMIT;
    END TRY
    BEGIN CATCH
        IF @tranCount = 0 ROLLBACK;
        THROW;
    END CATCH
END;
```
